class Solution {
    public int maxRotateFunction(int[] nums) {
        int n = nums.length;
        int totalSum = 0;
        int currentF = 0;
        
        for (int i = 0; i < n; i++) {
            totalSum += nums[i];
            currentF += i * nums[i];
        }
        
        int maxF = currentF;
        
        for (int k = 1; k < n; k++) {
            currentF = currentF + totalSum - n * nums[n - k];
            maxF = Math.max(maxF, currentF);
        }
        
        return maxF;
    }
}
