class Solution {
    public char[][] rotateTheBox(char[][] boxGrid) {
        int m = boxGrid.length;
        int n = boxGrid[0].length;
        
        // Step 1: Apply horizontal gravity row by row
        for (int i = 0; i < m; i++) {
            int emptySpot = n - 1;
            for (int j = n - 1; j >= 0; j--) {
                if (boxGrid[i][j] == '*') {
                    // Obstacle resets the empty spot to just before it
                    emptySpot = j - 1;
                } else if (boxGrid[i][j] == '#') {
                    // Move the stone to the furthest available empty spot
                    boxGrid[i][j] = '.';
                    boxGrid[i][emptySpot] = '#';
                    emptySpot--;
                }
            }
        }
        
        // Step 2: Rotate the grid 90 degrees clockwise
        char[][] rotatedBox = new char[n][m];
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                rotatedBox[j][m -1- i] = boxGrid[i][j];
            }
        }
        
        return rotatedBox;
    }
}
