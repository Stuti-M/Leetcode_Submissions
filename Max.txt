class Solution {
    public int maxPathScore(int[][] grid, int k) {
        int m = grid.length;
        int n = grid[0].length;
        int[][][] dp = new int[m][n][k + 1];
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                for (int c = 0; c <= k; c++) {
                    dp[i][j][c] = -1;
                }
            }
        }
        
        int[] startProps = getCostScore(grid[0][0]);
        if (startProps[0] <= k) {
            dp[0][0][startProps[0]] = startProps[1];
        }
        
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (i == 0 && j == 0) continue;
                
                int[] props = getCostScore(grid[i][j]);
                int currCost = props[0];
                int currScore = props[1];
                
                for (int c = currCost; c <= k; c++) {
                    int prevC = c - currCost;
                    int maxPrev = -1;
                    
                 
                    if (i > 0 && dp[i - 1][j][prevC] != -1) {
                        maxPrev = Math.max(maxPrev, dp[i - 1][j][prevC]);
                    }
               
                    if (j > 0 && dp[i][j - 1][prevC] != -1) {
                        maxPrev = Math.max(maxPrev, dp[i][j - 1][prevC]);
                    }
                    
                    if (maxPrev != -1) {
                        dp[i][j][c] = maxPrev + currScore;
                    }
                }
            }
        }
        

        int maxFinalScore = -1;
        for (int c = 0; c <= k; c++) {
            maxFinalScore = Math.max(maxFinalScore, dp[m - 1][n - 1][c]);
        }
        
        return maxFinalScore;
    }

    private int[] getCostScore(int val) {
        if (val == 0) return new int[]{0, 0};
        if (val == 1) return new int[]{1, 1};
        if (val == 2) return new int[]{1, 2};
        return new int[]{0, 0};
        
    }
}
