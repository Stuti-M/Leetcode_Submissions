class Solution {
    public String smallestPalindrome(String s) {
        int[] count = new int[26];
        for (char c : s.toCharArray()) {
            count[c - 'a']++;
        }
        
        StringBuilder left = new StringBuilder();
        String mid = "";
        
        // Build first half in alphabetical order ('a' to 'z')
        for (int i = 0; i < 26; i++) {
            char c = (char) ('a' + i);
            if (count[i] % 2 != 0) {
                mid = String.valueOf(c);
            }
            for (int j = 0; j < count[i] / 2; j++) {
                left.append(c);
            }
        }
        
        String firstHalf = left.toString();
        String rightHalf = left.reverse().toString();
        
        return firstHalf + mid + rightHalf;
    }
}
