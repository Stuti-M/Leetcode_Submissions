class Solution {
    private static final int MAX_K = 1_000_001;

    public String smallestPalindrome(String s, int k) {
        int n = s.length();
        int halfLen = n / 2;
        int[] cnt = new int[26];
        
        // Count character frequencies in the left half of the palindrome
        for (int i = 0; i < halfLen; i++) {
            cnt[s.charAt(i) - 'a']++;
        }
        
        char mid = (n % 2 == 1) ? s.charAt(halfLen) : 0;
        
        // Return empty string if total distinct permutations are less than k
        if (countArrangements(cnt) < k) {
            return "";
        }
        
        // Greedily build the left half character-by-character
        StringBuilder left = new StringBuilder();
        for (int pos = 0; pos < halfLen; pos++) {
            for (int ch = 0; ch < 26; ch++) {
                if (cnt[ch] == 0) continue;
                
                cnt[ch]--;
                int ways = countArrangements(cnt);
                
                if (k <= ways) {
                    left.append((char) ('a' + ch));
                    break;
                } else {
                    k -= ways;
                    cnt[ch]++; // Backtrack and try next character
                }
            }
        }
        
        // Construct final palindrome
        StringBuilder ans = new StringBuilder();
        ans.append(left);
        if (mid != 0) {
            ans.append(mid);
        }
        ans.append(new StringBuilder(left).reverse());
        return ans.toString();
    }
    
    // Calculates multinomial permutations truncated at MAX_K
    private int countArrangements(int[] cnt) {
        int total = 0;
        for (int c : cnt) total += c;
        
        long res = 1;
        for (int c : cnt) {
            if (c == 0) continue;
            res *= comb(total, c);
            if (res >= MAX_K) return MAX_K;
            total -= c;
        }
        return (int) res;
    }
    
    // Combination function nCr truncated at MAX_K
    private int comb(int n, int k) {
        if (k < 0 || k > n) return 0;
        k = Math.min(k, n - k);
        long res = 1;
        for (int i = 1; i <= k; i++) {
            res = res * (n - i + 1) / i;
            if (res >= MAX_K) return MAX_K;
        }
        return (int) res;
    }
}
